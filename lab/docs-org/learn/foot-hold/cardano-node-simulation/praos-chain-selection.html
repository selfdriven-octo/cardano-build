Algorithm Pseudocode
selectChain(current, candidate, k, stabilityWindow):
  
  // 1. Structural validation
  for block in candidate[1..]:
    assert block.prevHash == prev.hash
    assert block.slot > prev.slot
    assert vrfValid(block.vrfProof, block.issuer, block.slot)
    assert kesValid(block.kesSig, block.slot)
  
  // 2. Find fork point
  forkPoint = lastCommonBlock(current, candidate)
  rollbackDepth = current.length - indexOf(forkPoint)
  
  // 3. Security parameter check
  if rollbackDepth > k:
    return KEEP_CURRENT  // cannot roll back past k
  
  // 4. Length comparison (primary rule)
  if candidate.tip.blockNo > current.tip.blockNo:
    return SWITCH_TO_CANDIDATE
  if candidate.tip.blockNo < current.tip.blockNo:
    return KEEP_CURRENT
  
  // 5. Density tiebreak (equal length)
  window = [tip.slot - 3k/f .. tip.slot]
  if density(candidate, window) > density(current, window):
    return SWITCH_TO_CANDIDATE
  
  // 6. VRF tiebreak (equal density)
  if candidate.tip.vrfHash < current.tip.vrfHash:
    return SWITCH_TO_CANDIDATE
  
  return KEEP_CURRENT
