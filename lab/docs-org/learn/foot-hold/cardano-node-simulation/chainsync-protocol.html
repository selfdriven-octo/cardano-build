-- The ChainSync protocol as session types
-- (from ouroboros-network-protocols)

data ChainSync header point tip where

  StIdle :: ChainSync header point tip
    -- Consumer has agency

  StNext :: ChainSync header point tip
    -- Producer has agency (immediate response possible)

  StMustReply :: ChainSync header point tip
    -- Producer has agency (MUST eventually respond)

  StIntersect :: ChainSync header point tip
    -- Producer has agency

  StDone :: ChainSync header point tip
    -- Terminal state — nobody has agency

-- Messages are typed by their state transitions:

MsgRequestNext
  :: Message (ChainSync h p t) StIdle StNext

MsgAwaitReply
  :: Message (ChainSync h p t) StNext StMustReply

MsgRollForward  header tip
  :: Message (ChainSync h p t) StNext    StIdle
  |  Message (ChainSync h p t) StMustReply StIdle

MsgRollBackward point tip
  :: Message (ChainSync h p t) StNext    StIdle
  |  Message (ChainSync h p t) StMustReply StIdle

MsgFindIntersect [point]
  :: Message (ChainSync h p t) StIdle StIntersect

MsgIntersectFound    point tip
  :: Message (ChainSync h p t) StIntersect StIdle

MsgIntersectNotFound       tip
  :: Message (ChainSync h p t) StIntersect StIdle

MsgDone
  :: Message (ChainSync h p t) StIdle StDone
