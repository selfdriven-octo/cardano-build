-- The BlockFetch protocol as session types
-- (from ouroboros-network-protocols)

data BlockFetch block point where

  BFIdle :: BlockFetch block point
    -- Client has agency

  BFBusy :: BlockFetch block point
    -- Server has agency

  BFStreaming :: BlockFetch block point
    -- Server has agency (streaming blocks)

  BFDone :: BlockFetch block point
    -- Terminal state

-- Messages:

MsgRequestRange (ChainRange point)
  :: Message (BlockFetch b p) BFIdle BFBusy
  -- Client requests blocks in range [from..to]
  -- Points are (slot, hash) pairs

MsgStartBatch
  :: Message (BlockFetch b p) BFBusy BFStreaming
  -- Server has the range, will start sending

MsgNoBlocks
  :: Message (BlockFetch b p) BFBusy BFIdle
  -- Server doesn't have the requested range

MsgBlock block
  :: Message (BlockFetch b p) BFStreaming BFStreaming
  -- Server sends one block body

MsgBatchDone
  :: Message (BlockFetch b p) BFStreaming BFIdle
  -- Server finished sending all blocks in range

MsgClientDone
  :: Message (BlockFetch b p) BFIdle BFDone
  -- Client terminates the protocol

-- Key difference from ChainSync:
-- BlockFetch is RANGE-BASED, not cursor-based.
-- Client requests specific (from, to) ranges.
-- Server streams all blocks in that range.
-- Multiple ranges can be PIPELINED.
