-- KeepAlive is the simplest Ouroboros mini-protocol
-- Just 3 states, 3 messages

data KeepAlive where
  StClient :: KeepAlive    -- Client has agency
  StServer :: KeepAlive    -- Server has agency
  StDone   :: KeepAlive    -- Terminal

MsgKeepAlive (cookie :: Word16)
  :: Message KeepAlive StClient StServer
  -- Client sends ping with a 16-bit cookie
  -- Cookie is echoed back to match responses

MsgKeepAliveResponse (cookie :: Word16)
  :: Message KeepAlive StServer StClient
  -- Server echoes the cookie back
  -- Client measures RTT from send to receive

MsgDone
  :: Message KeepAlive StClient StDone
  -- Client terminates

-- Wire format (CBOR):
-- MsgKeepAlive:         [0, cookie]
-- MsgKeepAliveResponse: [1, cookie]
-- MsgDone:              [2]
--
-- Protocol ID on multiplexer: 8 (node-to-node)
-- Ping interval: configurable, default ~10s
-- Timeout: if no pong within ~timeout, disconnect
